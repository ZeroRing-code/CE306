const express = require("express");

const MAX_BIO_LENGTH = 500;
const MAX_AVATAR_URL_LENGTH = 2048;


// =====================================
// HELPER: ตรวจสอบ AVATAR URL
// =====================================

function validateAvatarUrl(value) {
    if (value === null || value === undefined || value === "") {
        return {
            valid: true,
            value: null
        };
    }

    if (typeof value !== "string") {
        return {
            valid: false,
            value: null
        };
    }

    const avatarUrl = value.trim();

    if (!avatarUrl) {
        return {
            valid: true,
            value: null
        };
    }

    if (avatarUrl.length > MAX_AVATAR_URL_LENGTH) {
        return {
            valid: false,
            value: null
        };
    }

    // อนุญาต URL ภายในเว็บไซต์ เช่น /uploads/avatar.png
    if (
        avatarUrl.startsWith("/") &&
        !avatarUrl.startsWith("//") &&
        !avatarUrl.includes("\\")
    ) {
        return {
            valid: true,
            value: avatarUrl
        };
    }

    try {
        const url = new URL(avatarUrl);

        if (
            url.protocol !== "http:" &&
            url.protocol !== "https:"
        ) {
            return {
                valid: false,
                value: null
            };
        }

        return {
            valid: true,
            value: avatarUrl
        };

    } catch (error) {
        return {
            valid: false,
            value: null
        };
    }
}

router.get("/me", requireAuth, async (req, res) => {
    try {

            `SELECT
                user_id,
                username,
                display_name,
                bio,
                avatar_url,
                created_at,
                updated_at
             FROM users

        if (!result.rowCount) {
            return res.status(404).json({
                message: "ไม่พบข้อมูลผู้ใช้งาน"
            });
        }

        res.json({
            profile: result.rows[0]
        });

    } catch (error) {
        console.error("Get My Profile:", error);

        res.status(500).json({
            message: "โหลดโปรไฟล์ไม่สำเร็จ"
        });
    }
});


// =====================================
// PUT: แก้ไขโปรไฟล์ของตัวเอง
// =====================================

router.put("/me", requireAuth, async (req, res) => {
    try {
        // โหลดข้อมูลเดิมก่อน เพื่อไม่ให้ข้อมูลหาย
        // กรณีไม่ได้ส่งบางฟิลด์เข้ามา
        const current = await pool.query(
            `SELECT
                display_name,
                bio,
                avatar_url
             FROM users
             WHERE user_id = $1`,
            [req.user.user_id]
        );

        if (!current.rowCount) {
            return res.status(404).json({
                message: "ไม่พบข้อมูลผู้ใช้งาน"
            });
        }

        const oldProfile = current.rows[0];

        const displayName =
            req.body.display_name === undefined
                ? oldProfile.display_name
                : typeof req.body.display_name === "string"
                    ? req.body.display_name.trim()
                    : null;

        const bio =
            req.body.bio === undefined
                ? oldProfile.bio
                : typeof req.body.bio === "string"
                    ? req.body.bio.trim() || null
                    : req.body.bio === null
                        ? null
                        : undefined;

        const rawAvatar =
            req.body.avatar_url === undefined
                ? oldProfile.avatar_url
                : req.body.avatar_url;

        // ตรวจสอบ Display Name
        if (
            typeof displayName !== "string" ||
            displayName.length < 1 ||
            displayName.length > 50
        ) {
            return res.status(400).json({
                message: "Display Name ต้องมีความยาว 1–50 ตัวอักษร"
            });
        }

        // ตรวจสอบ Bio
        if (
            bio !== null &&
            (
                typeof bio !== "string" ||
                bio.length > MAX_BIO_LENGTH
            )
        ) {
            return res.status(400).json({
                message: `Bio ต้องไม่เกิน ${MAX_BIO_LENGTH} ตัวอักษร`
            });
        }

        // ตรวจสอบ Avatar URL
        const avatar = validateAvatarUrl(rawAvatar);

        if (!avatar.valid) {
            return res.status(400).json({
                message: "URL รูปโปรไฟล์ไม่ถูกต้อง"
            });
        }

        // บันทึกข้อมูล
        const result = await pool.query(
            `UPDATE users
             SET
                display_name = $1,
                bio = $2,
                avatar_url = $3,
                updated_at = CURRENT_TIMESTAMP
             WHERE user_id = $4
             RETURNING
                user_id,
                username,
                display_name,
                bio,
                avatar_url,
                created_at,
                updated_at`,
            [
                displayName,
                bio,
                avatar.value,
                req.user.user_id
            ]
        );

        res.json({
            message: "อัปเดตโปรไฟล์สำเร็จ",
            profile: result.rows[0]
        });

    } catch (error) {
        console.error("Update Profile:", error);

        res.status(500).json({
            message: "อัปเดตโปรไฟล์ไม่สำเร็จ"
        });
    }
});

router.get("/:username", async (req, res) => {
    const { username } = req.params;

    if (
        typeof username !== "string" ||
        username.length < 1 ||
        username.length > 30
    ) {
        return res.status(400).json({
            message: "Username ไม่ถูกต้อง"
        });
    }

    try {
        // โหลดข้อมูลโปรไฟล์
        // ไม่เปิดเผย Email หรือ Password
        const userResult = await pool.query(
            `SELECT
                user_id,
                username,
                display_name,
                bio,
                avatar_url,
                created_at
             FROM users
             WHERE username = $1
               AND account_status = 'active'`,
            [username]
        );

        if (!userResult.rowCount) {
            return res.status(404).json({
                message: "ไม่พบผู้ใช้งานนี้"
            });
        }

        const user = userResult.rows[0];

        // โหลดนิยายที่มีตอนเผยแพร่แล้วเท่านั้น
        const novelsResult = await pool.query(
            `SELECT
                n.novel_id,
                n.title,
                n.description,
                n.cover_image_url,
                n.status,
                n.view_count,
                COUNT(c.chapter_id)::int AS chapter_count,
                MAX(c.published_at) AS last_published_at
             FROM novels n

             JOIN chapters c
                ON c.novel_id = n.novel_id
               AND c.status = 'published'

             WHERE n.user_id = $1

             GROUP BY
                n.novel_id,
                n.title,
                n.description,
                n.cover_image_url,
                n.status,
                n.view_count,
                n.updated_at

             ORDER BY
                MAX(c.published_at) DESC NULLS LAST,
                n.updated_at DESC`,
            [user.user_id]
        );

        res.json({
            profile: {
                ...user,
                novel_count: novelsResult.rowCount
            },
            novels: novelsResult.rows
        });

    } catch (error) {
        console.error("Get Public Profile:", error);

        res.status(500).json({
            message: "โหลดโปรไฟล์ไม่สำเร็จ"
        });
    }
});

async function api(url, options = {}) {
    const response = await fetch(url, {
        credentials: "same-origin",
        ...options
    });

    const data = await response.json().catch(() => ({}));

    if (!response.ok) {
        throw new Error(data.message || "เกิดข้อผิดพลาด");
    }

    return data;

async function initialize() {
    try {
        const account = await api("/api/auth/me");

        if (!account.authenticated) {
            loginRequired.hidden = false;
            return;
        }

        app.hidden = false;
        await loadGenres();

    } catch {
        loginRequired.hidden = false;
    }
}

async function loadGenres() {
    try {
        const genres = await api("/api/novels/genres");

        genreList.replaceChildren();

        genres.forEach(genre => {
            const label = document.createElement("label");
            label.className = "genre-option";

            const checkbox = document.createElement("input");
            checkbox.type = "checkbox";
            checkbox.name = "genres";
            checkbox.value = genre.genre_id;

            const text = document.createElement("span");
            text.textContent = genre.genre_name;

            label.append(checkbox, text);
            genreList.appendChild(label);
        });

    } catch (error) {
        genreList.textContent = error.message;
    }
}

// โหลด Genre ทั้งหมด
async function loadGenres(selectedIds = []) {
    const response = await api("/api/novels/genres");

    // รองรับทั้งข้อมูลแบบ Array และ Object
    const genres = Array.isArray(response)
        ? response
        : response.genres || [];

    if (!Array.isArray(genres)) {
        throw new Error("รูปแบบข้อมูล Genre ไม่ถูกต้อง");
    }

    genreList.replaceChildren();

    genres.forEach(genre => {
        const label = document.createElement("label");
        label.className = "genre-option";

        const checkbox = document.createElement("input");
        checkbox.type = "checkbox";
        checkbox.name = "genres";
        checkbox.value = genre.genre_id;

        checkbox.checked = selectedIds.some(
            id => Number(id) === Number(genre.genre_id)
        );

        const name = document.createElement("span");
        name.textContent = genre.genre_name;

        label.append(checkbox, name);
        genreList.appendChild(label);
    });
}

const form = document.getElementById("edit-profile-form");

// =====================================
// แสดงข้อความแจ้งเตือน
// =====================================

function showMessage(text, type = "") {
    message.textContent = text;
    message.className = `message ${type}`;
}


// =====================================
// จัดการรูปโปรไฟล์
// =====================================

function updateAvatar() {
    const displayName = displayNameInput.value.trim();

    avatarInitial.textContent =
        (displayName || currentUsername || "?").charAt(0).toUpperCase();

    const url = avatarUrlInput.value.trim();

    if (!url) {
        avatarImage.style.display = "none";
        avatarImage.removeAttribute("src");
        return;
    }

    avatarImage.style.display = "none";
    avatarImage.onload = () => {
        avatarImage.style.display = "block";
    };

    avatarImage.onerror = () => {
        avatarImage.style.display = "none";
    };

    avatarImage.src = url;
}


// =====================================
// โหลดข้อมูลโปรไฟล์
// =====================================

async function loadProfile() {
    try {
        const response = await fetch("/api/profile/me", {
            method: "GET",
            credentials: "same-origin"
        });

        const data = await response.json();

        if (!response.ok) {
            throw new Error(data.message || "ไม่สามารถโหลดข้อมูลโปรไฟล์ได้");
        }

        const profile = data.profile || data.user || data;

        currentUsername = profile.username || "";

        usernameInput.value = currentUsername;
        displayNameInput.value = profile.display_name || "";
        bioInput.value = profile.bio || "";
        avatarUrlInput.value = profile.avatar_url || "";

        bioCount.textContent = bioInput.value.length;

        backLink.href = currentUsername
            ? `/profile.html?username=${encodeURIComponent(currentUsername)}`
            : "/profile.html";

        cancelButton.addEventListener("click", () => {
            window.location.href = backLink.href;
        });

        updateAvatar();

    } catch (error) {
        showMessage(error.message, "error");

        if (error.message.toLowerCase().includes("unauthorized") ||
            error.message.includes("เข้าสู่ระบบ")) {
            window.location.href = "/login.html";
        }
    }
}

// =====================================
// ตรวจสอบ URL รูปภาพ
// =====================================

function isValidAvatarUrl(value) {
    if (!value) return true;

    if (value.startsWith("/")) {
        return true;
    }

    try {
        const url = new URL(value);
        return url.protocol === "http:" || url.protocol === "https:";
    } catch {
        return false;
    }
}


// =====================================
// บันทึกข้อมูลโปรไฟล์
// =====================================

form.addEventListener("submit", async (event) => {
    event.preventDefault();

    const displayName = displayNameInput.value.trim();
    const bio = bioInput.value.trim();
    const avatarUrl = avatarUrlInput.value.trim();

    if (!displayName) {
        showMessage("กรุณากรอกชื่อที่แสดง", "error");
        return;
    }

    if (displayName.length > 50) {
        showMessage("ชื่อที่แสดงต้องไม่เกิน 50 ตัวอักษร", "error");
        return;
    }

    if (bio.length > 500) {
        showMessage("ประวัติส่วนตัวต้องไม่เกิน 500 ตัวอักษร", "error");
        return;
    }

    if (!isValidAvatarUrl(avatarUrl)) {
        showMessage("URL รูปภาพต้องเป็น HTTP หรือ HTTPS", "error");
        return;
    }

    saveButton.disabled = true;
    saveButton.textContent = "กำลังบันทึก...";

    showMessage("");

    try {
        const response = await fetch("/api/profile/me", {
            method: "PUT",
            credentials: "same-origin",
            headers: {
                "Content-Type": "application/json"
            },
            body: JSON.stringify({
                display_name: displayName,
                bio: bio,
                avatar_url: avatarUrl
            })
        });

        const data = await response.json();

        if (!response.ok) {
            throw new Error(data.message || "ไม่สามารถบันทึกข้อมูลได้");
        }

        showMessage("บันทึกข้อมูลเรียบร้อยแล้ว", "success");

        setTimeout(() => {
            window.location.href = backLink.href;
        }, 1000);

    } catch (error) {
        showMessage(error.message, "error");

    } finally {
        saveButton.disabled = false;
        saveButton.textContent = "บันทึกข้อมูล";
    }
});
